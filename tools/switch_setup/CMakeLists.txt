cmake_minimum_required(VERSION 3.20)
project(moon_switch_setup LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "The Switch deployment window is a Windows application")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(moon_switch_setup WIN32 main.cpp)
target_compile_features(moon_switch_setup PRIVATE cxx_std_17)
target_compile_options(moon_switch_setup PRIVATE /utf-8)
target_compile_definitions(moon_switch_setup PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(moon_switch_setup PRIVATE user32 gdi32 comdlg32 shell32 ole32 propsys
  uuid wininet bcrypt ws2_32 comctl32)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(EXISTS "${REPO}/res/moonengine.ico")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/icon.rc" "1 ICON \"${REPO}/res/moonengine.ico\"\n")
  target_sources(moon_switch_setup PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/icon.rc")
endif()
set_target_properties(moon_switch_setup PROPERTIES OUTPUT_NAME "MoonEngine-Switch-Setup")
